if(NOT SHIFTWHEEL_UPDATE_PUBLIC_KEY STREQUAL "")
  string(LENGTH "${SHIFTWHEEL_UPDATE_PUBLIC_KEY}" _key_length)
  if(NOT _key_length EQUAL 144 OR NOT SHIFTWHEEL_UPDATE_PUBLIC_KEY MATCHES "^[0-9a-fA-F]+$" OR NOT SHIFTWHEEL_UPDATE_KEY_ID MATCHES "^[A-Za-z0-9_-]+$")
    message(FATAL_ERROR "Invalid pinned update public key or identifier")
  endif()
endif()
