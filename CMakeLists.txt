cmake_minimum_required(VERSION 3.24)
project(ShiftWheel VERSION 0.6.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32)
  message(FATAL_ERROR "ShiftWheel requires Windows")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(shiftwheel_core src/policy.cpp src/file_policy.cpp src/engines.cpp src/options.cpp src/progress.cpp)
target_include_directories(shiftwheel_core PUBLIC src)
add_library(shiftwheel_images src/image_converter.cpp)
target_include_directories(shiftwheel_images PUBLIC src)
target_link_libraries(shiftwheel_images PUBLIC shiftwheel_core windowscodecs ole32 oleaut32)
add_library(shiftwheel_ui src/wheel.cpp)
add_library(shiftwheel_panel src/control_panel.cpp)
target_link_libraries(shiftwheel_panel PUBLIC shiftwheel_core comctl32 shell32 ole32)
target_link_libraries(shiftwheel_ui PUBLIC shiftwheel_core d2d1 dwrite shell32 shcore windowscodecs ole32)
add_executable(ShiftWheel WIN32 src/main.cpp src/app.cpp src/worker.cpp resources/app.rc resources/app.manifest)
add_library(shiftwheel_conversion src/converter.cpp src/media_converter.cpp src/pdf_converter.cpp src/avif_converter.cpp src/webp_converter.cpp src/presentation_converter.cpp src/ocr.cpp src/heic_converter.cpp)
target_link_libraries(shiftwheel_conversion PUBLIC shiftwheel_images shlwapi)
target_link_libraries(ShiftWheel PRIVATE shiftwheel_conversion shiftwheel_ui shiftwheel_panel shell32)
add_library(shiftwheel_updates src/updates.cpp)
target_include_directories(shiftwheel_updates PUBLIC src)
target_link_libraries(shiftwheel_updates PUBLIC winhttp bcrypt advapi32 shell32 ole32)
target_link_libraries(ShiftWheel PRIVATE shiftwheel_updates)
include(CTest)
if(BUILD_TESTING)

  add_executable(updates_tests tests/updates_tests.cpp)

  add_library(shiftwheel_update_test_engine src/updates.cpp)
  target_compile_definitions(shiftwheel_update_test_engine PRIVATE SHIFTWHEEL_UPDATE_TESTING)
  target_include_directories(shiftwheel_update_test_engine PUBLIC src)

  set_tests_properties(media_gpu_policy PROPERTIES TIMEOUT 30)

    target_link_libraries(${check}_tests PRIVATE shiftwheel_core)

    set_tests_properties(${check} PROPERTIES TIMEOUT 30)
  endforeach()
  add_executable(panel_tests tests/panel_tests.cpp resources/app.manifest)

  add_test(NAME worker_protocol COMMAND worker_protocol_tests)
  set_tests_properties(worker_protocol PROPERTIES TIMEOUT 30)

  set_tests_properties(application_smoke worker_smoke PROPERTIES TIMEOUT 30)
  set(SHIFTWHEEL_TEST_PYTHON "" CACHE FILEPATH "Python with reportlab, pypdf, python-docx, xlsxwriter and Pillow for real engine tests")
  if(SHIFTWHEEL_TEST_PYTHON)
    foreach(group pdf office media avif formats)
